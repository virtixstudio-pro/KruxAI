match /krux_quotas/{uid} {
  allow read: if request.auth != null && request.auth.uid == uid;
  // une création d'app à la fois, jamais de baisse du compteur dans la même semaine
  allow create: if request.auth != null && request.auth.uid == uid
                && request.resource.data.used == 1
                && request.resource.data.week.matches('^20[0-9]{2}-W[0-9]{1,2}$');
  allow update: if request.auth != null && request.auth.uid == uid
                && request.resource.data.week.matches('^20[0-9]{2}-W[0-9]{1,2}$')
                && ((request.resource.data.week == resource.data.week && request.resource.data.used == resource.data.used + 1)
                 || (request.resource.data.week != resource.data.week && request.resource.data.used == 1));
}

match /users/{uid}/apps/{appId} {
  allow read, create: if request.auth != null && request.auth.uid == uid;
}

// Limite connue : ces règles bloquent les erreurs et les retours à zéro simples, mais une app modifiée pourrait
// encore envoyer un faux nom de semaine. Une vraie garantie demande un serveur (Cloud Function / serveur Virtix Studio).
